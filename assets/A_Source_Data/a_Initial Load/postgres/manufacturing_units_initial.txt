DROP TABLE IF EXISTS manufacturing_units;

CREATE TABLE manufacturing_units (
    plant_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    plant_name VARCHAR(100) NOT NULL,
    unit_type VARCHAR(30) NOT NULL CHECK (unit_type IN ('COMPONENT_FABRICATION', 'FINAL_ASSEMBLY')), -- NEW LOGIC FLAG
    region VARCHAR(50),
    country VARCHAR(50) NOT NULL,
    scrap_cost_factor DECIMAL(5,2) DEFAULT 1.00,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

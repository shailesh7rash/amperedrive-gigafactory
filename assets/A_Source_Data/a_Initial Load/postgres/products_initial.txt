DROP TABLE IF EXISTS products;

-- 1. Create the finalized products dimension table
CREATE TABLE products (
    product_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    plant_id INT REFERENCES manufacturing_units(plant_id),
    product_name VARCHAR(100) NOT NULL,
    product_category VARCHAR(50) NOT NULL,
    material_cost_usd DECIMAL(10,2) NOT NULL,
    factory_output_cost_usd DECIMAL(10,2) NOT NULL,
    standard_cycle_time_secs INT NOT NULL,
    inserted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

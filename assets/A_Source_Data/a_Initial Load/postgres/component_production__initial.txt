DROP TABLE IF EXISTS component_production;


-- Create the component production table and insert data
CREATE TABLE component_production (
    component_run_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    plant_id INT REFERENCES manufacturing_units(plant_id),
    product_id INT REFERENCES products(product_id),
    equipment_id INT, -- Logically maps to Azure Storage Equipment CSV ID
    start_time TIMESTAMP NOT NULL,
    end_time TIMESTAMP,
    quantity_produced INT DEFAULT 0,
    quantity_rejected INT DEFAULT 0,
    downtime_minutes INT DEFAULT 0,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

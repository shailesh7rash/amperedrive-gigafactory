-- 3. Execute Idempotent Merge targeting the production warehouse table
INSERT INTO component_production (
    component_run_id, plant_id, product_id, equipment_id, 
    start_time, end_time, quantity_produced, quantity_rejected, 
    downtime_minutes, updated_at
)
OVERRIDING SYSTEM VALUE
SELECT 
    component_run_id, plant_id, product_id, equipment_id, 
    start_time, end_time, quantity_produced, quantity_rejected, 
    downtime_minutes, last_modified_at
FROM staging_component_production_delta
ON CONFLICT (component_run_id) 
DO UPDATE SET 
    end_time = EXCLUDED.end_time,
    quantity_produced = EXCLUDED.quantity_produced,
    quantity_rejected = EXCLUDED.quantity_rejected,
    downtime_minutes = EXCLUDED.downtime_minutes,
    updated_at = EXCLUDED.updated_at;

-- 4. Sync the identity sequence so future auto-inserts don't conflict
SELECT setval(
    pg_get_serial_sequence('component_production', 'component_run_id'), 
    COALESCE((SELECT MAX(component_run_id) FROM component_production), 1)
);

-- 5. Clean workspace
DROP TABLE IF EXISTS staging_component_production_delta;

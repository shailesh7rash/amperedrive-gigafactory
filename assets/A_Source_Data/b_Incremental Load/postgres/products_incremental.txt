-- 3. Execute the standard idempotent Upsert logic into the target warehouse table
INSERT INTO products (
    product_id, plant_id, product_name, product_category, 
    material_cost_usd, factory_output_cost_usd, standard_cycle_time_secs, updated_at
)
OVERRIDING SYSTEM VALUE
SELECT 
    product_id, plant_id, product_name, product_category, 
    material_cost_usd, factory_output_cost_usd, standard_cycle_time_secs, last_modified_at
FROM staging_products_delta
ON CONFLICT (product_id) DO UPDATE SET
    material_cost_usd = EXCLUDED.material_cost_usd,
    factory_output_cost_usd = EXCLUDED.factory_output_cost_usd,
    standard_cycle_time_secs = EXCLUDED.standard_cycle_time_secs,
    updated_at = EXCLUDED.updated_at;

-- 4. Sync the identity sequence so future auto-inserts don't conflict
SELECT setval(
    pg_get_serial_sequence('products', 'product_id'), 
    COALESCE((SELECT MAX(product_id) FROM products), 1)
);
